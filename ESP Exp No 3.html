Experiment Details

Experiment: ESP32-CAM Experiment No. 3

Experiment Name: Web Controlled Camera with Live Streaming

Controller: ESP32-CAM

Communication: WiFi

Web Server: ESP32 WebServer

Output: Live Camera Stream

Additional Functions: Capture Photo, Flash ON/OFF, Video Size Selection

Programming Software: Arduino IDE

Objective: To create a web-based camera interface using ESP32-CAM that provides live video streaming, photo capture, flash control, and camera resolution control through a web browser.

Required Components

ESP32-CAM

Main controller used to capture and stream camera images.

OV2640 Camera

Captures images and provides the video stream.

WiFi Network

Provides wireless connection for accessing the web interface.

Computer / Smartphone

Used to access and control the ESP32-CAM web page.

Circuit Diagram

The ESP32-CAM camera module is connected directly to the ESP32-CAM board using its camera interface. GPIO 4 is used to control the flash LED.

Working Principle

The ESP32-CAM is configured as a web server and connected to a WiFi network.

When a user opens the ESP32-CAM IP address in a browser, the ESP32 sends an HTML web page containing the camera controls.

The /stream route sends camera frames continuously using an MJPEG-style multipart stream.

The Capture button requests a high-resolution image from the camera.

The Flash ON and Flash OFF buttons control the flash LED connected to GPIO 4.

The VGA and QVGA buttons change the camera frame size through the /size web request.

esp32_cam_web_camera.ino 
#include "esp_camera.h"

#include <WiFi.h>

const char *ssid =
"Simusoft_Technologies";

"YOUR_WIFI_PASSWORD";

// ---------- SETUP FUNCTION ----------

void setup() {

    Serial.begin(115200);

    pinMode(4, OUTPUT);

    digitalWrite(4, LOW);

    camera_config_t config;

    config.ledc_channel =
        LEDC_CHANNEL_0;

    config.ledc_timer =
        LEDC_TIMER_0;

    config.pin_d0 =
        Y2_GPIO_NUM;

    config.pin_d1 =
        Y3_GPIO_NUM;

    config.pin_d2 =
        Y4_GPIO_NUM;

    config.pin_d3 =
        Y5_GPIO_NUM;

    config.pin_d4 =
        Y6_GPIO_NUM;

    config.pin_d5 =
        Y7_GPIO_NUM;

    config.pin_d6 =
        Y8_GPIO_NUM;

    config.pin_d7 =
        Y9_GPIO_NUM;

    config.pin_xclk =
        XCLK_GPIO_NUM;

    config.pin_pclk =
        PCLK_GPIO_NUM;

    config.pin_vsync =
        VSYNC_GPIO_NUM;

    config.pin_href =
        HREF_GPIO_NUM;

    config.pin_sccb_sda =
        SIOD_GPIO_NUM;

    config.pin_sccb_scl =
        SIOC_GPIO_NUM;

    config.pin_pwdn =
        PWDN_GPIO_NUM;

    config.pin_reset =
        RESET_GPIO_NUM;

    config.xclk_freq_hz =
        20000000;

    config.pixel_format =
        PIXFORMAT_JPEG;

    config.frame_size =
        FRAMESIZE_QVGA;

    config.jpeg_quality =
        15;

    config.fb_count =
        2;

    config.grab_mode =
        CAMERA_GRAB_LATEST;

    if (esp_camera_init(&config)
        != ESP_OK) {

        Serial.println(
            "Camera init failed"
        );

        return;

    }

    sensor_t *s =
        esp_camera_sensor_get();

    s->set_vflip(s, 1);

    WiFi.begin(
        ssid,
        password
    );

    while (
        WiFi.status() != WL_CONNECTED
    ) {

        delay(300);

    }

    WiFi.setSleep(false);

    Serial.print("IP: ");

    Serial.println(
        WiFi.localIP()
    );

    server.on("/", []() {

        server.send_P(
            200,
            "text/html",
            HTML_PAGE
        );

    });

    server.on(
        "/stream",
        HTTP_GET,
        []() {

            streamClient =
                server.client();

            streamClient.println(
                "HTTP/1.1 200 OK"
            );

            streamClient.println(
                "Content-Type: multipart/x-mixed-replace; boundary=frame"
            );

            streamClient.println();

        }
    );

    server.on(
        "/capture",
        HTTP_GET,
        []() {

            sensor_t *s =
                esp_camera_sensor_get();

            s->set_framesize(
                s,
                FRAMESIZE_VGA
            );

            camera_fb_t *fb =
                esp_camera_fb_get();

            if (!fb) {

                server.send(
                    500,
                    "text/plain",
                    "Camera error"
                );

                return;

            }

            WiFiClient client =
                server.client();

            client.println(
                "HTTP/1.1 200 OK"
            );

            client.println(
                "Content-Type: image/jpeg"
            );

            client.printf(
                "Content-Length: %u\r\n\r\n",
                fb->len
            );

            client.write(
                fb->buf,
                fb->len
            );

            esp_camera_fb_return(fb);

            client.stop();

            s->set_framesize(
                s,
                FRAMESIZE_QVGA
            );

        }
    );

    server.on(
        "/flash_on",
        []() {

            digitalWrite(
                4,
                HIGH
            );

            server.send(
                200,
                "text/plain",
                "OK"
            );

        }
    );

    server.on(
        "/flash_off",
        []() {

            digitalWrite(
                4,
                LOW
            );

            server.send(
                200,
                "text/plain",
                "OK"
            );

        }
    );

    server.on(
        "/size",
        []() {

            if (
                !server.hasArg("v")
            ) {

                server.send(
                    400,
                    "text/plain",
                    "Missing arg"
                );

                return;

            }

            sensor_t *s =
                esp_camera_sensor_get();

            s->set_framesize(
                s,
                (framesize_t)
                server.arg("v").toInt()
            );

            server.send(
                200,
                "text/plain",
                "OK"
            );

        }
    );

    server.begin();

}

// ---------- LOOP FUNCTION ----------

void loop() {

    server.handleClient();

    if (
        streamClient &&
        streamClient.connected()
    ) {

        camera_fb_t *fb =
            esp_camera_fb_get();

        if (!fb) return;

        streamClient.printf(

            "--frame\r\n"
            "Content-Type: image/jpeg\r\n"
            "Content-Length: %u\r\n\r\n",

            fb->len

        );

        streamClient.write(
            fb->buf,
            fb->len
        );

        streamClient.print(
            "\r\n"
        );

        esp_camera_fb_return(fb);

        delay(20);

    }

}

Steps to Perform the Experiment

  1. Connect the ESP32-CAM to the computer using a suitable programming interface.
  2. Open Arduino IDE.
  3. Select the appropriate ESP32-CAM board.
  4. Enter the WiFi SSID and password in the program.
  5. Upload the program to the ESP32-CAM.
  6. Open the Serial Monitor and set the baud rate to 115200 .
  7. Wait for the ESP32-CAM to connect to the WiFi network.
  8. Note the IP address displayed in the Serial Monitor.
  9. Open the IP address in a browser on a device connected to the same WiFi network.
  10. Observe the live ESP32-CAM video stream.
  11. Click Capture to capture a photo.
  12. Use Flash ON and Flash OFF to control the flash LED.
  13. Use VGA or QVGA to change the video size.

Expected Output

The ESP32-CAM connects successfully to the configured WiFi network and displays its local IP address in the Serial Monitor.

When the IP address is opened in a browser, a web page titled ESP32-CAM Live Stream is displayed.

The live camera stream is shown on the webpage.

The Capture button captures a photograph.

The Flash ON and Flash OFF buttons control the flash LED.

The VGA and QVGA buttons change the camera frame size.
